cmake_minimum_required(VERSION 3.22.1)
project(sumh_observer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_PLATFORM_LEVEL LESS 31)
    message(FATAL_ERROR "Observer artifacts require Android arm64-v8a API 31+")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include(FetchContent)
FetchContent_Declare(mbedtls
    URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.2/mbedtls-3.6.2.tar.bz2
    URL_HASH SHA256=8b54fb9bcf4d5a7078028e0520acddefb7900b3e66fec7f7175bb5b7d85ccdca)
FetchContent_GetProperties(mbedtls)
if(NOT mbedtls_POPULATED)
    FetchContent_Populate(mbedtls)
endif()
add_library(sumh_observer_sha256 STATIC
    "${mbedtls_SOURCE_DIR}/library/sha256.c"
    "${mbedtls_SOURCE_DIR}/library/platform_util.c")
target_include_directories(sumh_observer_sha256 PUBLIC "${mbedtls_SOURCE_DIR}/include" "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(sumh_observer_sha256 PUBLIC MBEDTLS_CONFIG_FILE="observer_mbedtls_config.h")
set(SOURCE_ID_INPUT "sumh-observer-v1;mbedtls-3.6.2;8b54fb9bcf4d5a7078028e0520acddefb7900b3e66fec7f7175bb5b7d85ccdca;")
set(IDENTITY_FILES CMakeLists.txt collector.cpp collector.hpp protocol.cpp protocol.hpp
    main.cpp jni.cpp observer_mbedtls_config.h ../../../userspace/common/json.hpp ../../../userspace/common/utf8.hpp)
foreach(source IN LISTS IDENTITY_FILES)
    if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/${source}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/${source}")
        file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/${source}" source_hash)
        string(APPEND SOURCE_ID_INPUT "${source}:${source_hash};")
    else()
        message(FATAL_ERROR "Source identity input missing: ${source}")
    endif()
endforeach()
string(SHA256 OBSERVER_SOURCE_ID "${SOURCE_ID_INPUT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/collector-build-id.txt" "sha256-v1:${OBSERVER_SOURCE_ID}\n")
add_library(sumh_observer_core STATIC collector.cpp protocol.cpp)
target_include_directories(sumh_observer_core PUBLIC "${REPO_ROOT}/userspace/common" "${CMAKE_CURRENT_LIST_DIR}")
target_compile_options(sumh_observer_core PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(sumh_observer_core PRIVATE SUMH_OBSERVER_BUILD_ID="sha256-v1:${OBSERVER_SOURCE_ID}")
target_link_libraries(sumh_observer_core PUBLIC sumh_observer_sha256)
add_executable(sumh-observer main.cpp)
target_link_libraries(sumh-observer PRIVATE sumh_observer_core)
target_compile_options(sumh-observer PRIVATE -Wall -Wextra -Werror)
add_library(sumh_observer_jni SHARED jni.cpp)
target_link_libraries(sumh_observer_jni PRIVATE sumh_observer_core)
target_compile_options(sumh_observer_jni PRIVATE -Wall -Wextra -Werror)
